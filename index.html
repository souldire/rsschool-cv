Пример кода

Решение задачи «Find the odd int» (6 kyu) с Codewars:

function findOdd(A) {
  const counts = {};
  for (const n of A) {
    counts[n] = (counts[n] || 0) + 1;
  }
  for (const n in counts) {
    if (counts[n] % 2 === 1) {
      return Number(n);
    }
  }
  return 0;
}
